7// Observer pattern shared by the three parts (Tower, Map, Critter).
8//
9// Observable (the subject) keeps a list of observers and calls notify() every time one of its values changes.
10// Observer is the interface of the views: update(subject) is called by notify(), and the view then reads the
11// new values from the subject itself (pull model), so the subject never needs to know how it is displayed.
12//
13// Header only, so every program that already includes Map.h, Tower.h or Critter.h builds without new source files.
14
15class Observable;

17// Interface of every view that wants to be told when an Observable changes.
18class Observer {
19 public:
20 virtual ~Observer() = default;
21
22 // Called by Observable::notify() each time a value of the subject changed.
23 virtual void update(Observable* subject) = 0;

25 // Called when the subject is destroyed while this observer is still attached, so the observer can forget it
26 // instead of keeping a dangling pointer. Does nothing by default.
27 virtual void subjectDestroyed(Observable* subject) { (void)subject; }

31class Observable {

36 // Tells every attached observer that a value changed. Called by the subclasses after each change.
37 void notify() {
38 // iterate on a copy: an observer may detach itself (or another one) while it is being notified.
39 std::vector<Observer*> current = observers;
40 for (Observer* observer : current) {
41 if (std::find(observers.begin(), observers.end(), observer) != observers.end()) {
42 observer->update(this);
43 }
44 }
45 }
46
47 public:
48 Observable() = default;
49
50 // A copy starts with no observers: the observers watch one object, not its values.
51 Observable(const Observable&) {}
52
53 // Assigning new values keeps this object's own observers (the copied object's observers are not taken).
54 // The subclass is responsible for calling notify() when it assigns new values on purpose (see Map::loadFromFile).
55 Observable& operator=(const Observable&) { return *this; }
56
57 virtual ~Observable() {
58 std::vector<Observer*> current = observers;
59 observers.clear();
60 for (Observer* observer : current) {
61 observer->subjectDestroyed(this);
62 }
63 }
64
65 // Plugs an observer. Attaching the same observer twice or a null pointer does nothing.
66 void attach(Observer* observer) {
67 if (observer != nullptr && std::find(observers.begin(), observers.end(), observer) == observers.end()) {
68 observers.push_back(observer);
69 }
70 }
71
72 // Unplugs an observer. Does nothing if it was not attached.
73 void detach(Observer* observer) {
74 observers.erase(std::remove(observers.begin(), observers.end(), observer), observers.end());
75 }
76
77 int getObserverCount() const {
